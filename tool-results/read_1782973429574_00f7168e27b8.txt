    95→def group_videos_by_variant(videos: list[Video], *, merge: bool) -> list[list[Video]]:
    96→    """Group video tracks for muxing.
    97→
    98→    When ``merge`` is True, tracks sharing ``(height, range, codec)`` are grouped into one
    99→    file so only language varies within a group; different resolutions, ranges and codecs
   100→    stay in separate groups (separate files). When False, each track is its own group
   101→    (one file per track, the default behaviour). Group order follows first-seen track order.
   102→    """
   103→    if not merge:
   104→        return [[video] for video in videos]
   105→    groups: dict[tuple[Any, ...], list[Video]] = {}
   106→    for video in videos:
   107→        groups.setdefault((video.height, video.range, video.codec), []).append(video)
   108→    return list(groups.values())

   111→def apply_service_dl_overrides(ctx: click.Context, service_dl_config: dict[str, Any], log: logging.Logger) -> None:
   112→    """Apply ``services.<TAG>.dl`` config onto ``ctx.params``. Explicit CLI/env values win;
   113→    defaults and global ``dl:`` default_map values are replaced."""
   114→    params_by_name = {param.name: param for param in ctx.command.params if param.name}
   115→    for name, value in normalize_dl_config(service_dl_config).items():
   116→        param = params_by_name.get(name)
   117→        if param is None:
   118→            log.warning(f"Ignoring unknown dl option '{name}' in service config")
   119→            continue
   120→        if name not in ctx.params:
   121→            log.debug(f"Skipping service dl override '{name}': not present in this context")
   122→            continue
   123→        if value is None:
   124→            continue
   125→        source = ctx.get_parameter_source(name)
   126→        if source not in (ParameterSource.DEFAULT, ParameterSource.DEFAULT_MAP):
   127→            continue
   128→        try:
   129→            ctx.params[name] = param.type_cast_value(ctx, value)
   130→            log.debug(f"Applied service dl override '{name}': {ctx.params[name]}")
   131→        except Exception as e:
   132→            log.warning(
   133→                f"Failed to apply service dl override '{name}'={value!r}: {e}. "
   134→                f"Check that the value is valid for this parameter."
   135→            )

   138→def download_tracks_in_passes(
   139→    tracks: Iterable[AnyTrack],
   140→    max_concurrent: int,
   141→    run_one: Callable[[AnyTrack, int], None],
   142→    *,
   143→    skip_subtitle_errors: bool,
   144→    on_subtitle_skipped: Callable[[Subtitle], None],
   145→) -> None:
   146→    """Download a title's tracks so a skippable subtitle failure can't corrupt the rest.
   147→
   148→    A failed track sets the process-global ``DOWNLOAD_CANCELLED`` event, making other in-flight
   149→    tracks early-return without raising. With ``skip_subtitle_errors`` set, video/audio download
   150→    concurrently first; the subtitles then run one at a time (a concurrent pass would let one
   151→    failure's cancel silently drop the others, unrecorded), with the event cleared before each.
   152→    Video/audio failures stay fatal. The event is cleared on entry (stale cancel from a prior
   153→    title) and in ``finally`` (never leave it set for later code).
   154→
   155→    ``run_one(track, index)`` downloads a single track; ``on_subtitle_skipped(track)`` records a
   156→    subtitle whose download raised (handled here).
   157→    """
   158→    DOWNLOAD_CANCELLED.clear()
   159→    indexed = list(enumerate(tracks))
   160→    if skip_subtitle_errors:
   161→        primary = [(i, t) for i, t in indexed if not isinstance(t, Subtitle)]
   162→        skippable = [(i, t) for i, t in indexed if isinstance(t, Subtitle)]
   163→    else:
   164→        primary, skippable = indexed, []
   165→
   166→    try:
   167→        with ThreadPoolExecutor(max_concurrent) as pool:
   168→            future_to_track = {pool.submit(run_one, track, i): track for i, track in primary}
   169→            for download in futures.as_completed(future_to_track):
   170→                download.result()  # a video/audio failure is fatal
   171→
   172→        for i, track in skippable:
   173→            DOWNLOAD_CANCELLED.clear()
   174→            try:
   175→                run_one(track, i)
   176→            except Exception:
   177→                on_subtitle_skipped(track)
   178→    finally:
   179→        DOWNLOAD_CANCELLED.clear()  # never leave a failed track's cancel set for later code

   195→    def find_custom_font(self, font_name: str) -> Optional[Path]:
   196→        """
   197→        Find font in custom fonts directory.
   198→
   199→        Args:
   200→            font_name: Font family name to find
   201→
   202→        Returns:
   203→            Path to font file, or None if not found
   204→        """
   205→        family_dir = Path(config.directories.fonts, font_name)
   206→        if family_dir.exists():
   207→            fonts = list(family_dir.glob("*.*tf"))
   208→            return fonts[0] if fonts else None
   209→        return None

   211→    def prepare_temp_font(
   212→        self, font_name: str, matched_font: Path, system_fonts: dict[str, Path], temp_font_files: list[Path]
   213→    ) -> Path:
   214→        """
   215→        Copy system font to temp and log if using fallback.
   216→
   217→        Args:
   218→            font_name: Requested font name
   219→            matched_font: Path to matched system font
   220→            system_fonts: Dictionary of available system fonts
   221→            temp_font_files: List to track temp files for cleanup
   222→
   223→        Returns:
   224→            Path to temp font file
   225→        """
   226→        # Find the matched name for logging
   227→        matched_name = next((name for name, path in system_fonts.items() if path == matched_font), None)
   228→
   229→        if matched_name and matched_name.lower() != font_name.lower():
   230→            self.log.info(f"Using '{matched_name}' as fallback for '{font_name}'")
   231→
   232→        # Create unique temp file path
   233→        safe_name = font_name.replace(" ", "_").replace("/", "_")
   234→        temp_path = config.directories.temp / f"font_{safe_name}{matched_font.suffix}"
   235→
   236→        # Copy if not already exists
   237→        if not temp_path.exists():
   238→            shutil.copy2(matched_font, temp_path)
   239→            temp_font_files.append(temp_path)
   240→
   241→        return temp_path

   243→    def attach_subtitle_fonts(
   244→        self, font_names: list[str], title: Title_T, temp_font_files: list[Path]
   245→    ) -> tuple[int, list[str]]:
   246→        """
   247→        Attach fonts for subtitle rendering.
   248→
   249→        Args:
   250→            font_names: List of font names requested by subtitles
   251→            title: Title object to attach fonts to
   252→            temp_font_files: List to track temp files for cleanup
   253→
   254→        Returns:
   255→            Tuple of (fonts_attached_count, missing_fonts_list)
   256→        """
   257→        system_fonts = get_system_fonts()
   258→
   259→        font_count = 0
   260→        missing_fonts = []
   261→
   262→        for font_name in set(font_names):
   263→            # Try custom fonts first
   264→            if custom_font := self.find_custom_font(font_name):
   265→                title.tracks.add(Attachment(path=custom_font, name=f"{font_name} ({custom_font.stem})"))
   266→                font_count += 1
   267→                continue
   268→
   269→            # Try system fonts with fallback
   270→            if system_font := find_font_with_fallbacks(font_name, system_fonts):
   271→                temp_path = self.prepare_temp_font(font_name, system_font, system_fonts, temp_font_files)
   272→                title.tracks.add(Attachment(path=temp_path, name=f"{font_name} ({system_font.stem})"))
   273→                font_count += 1
   274→            else:
   275→                self.log.warning(f"Subtitle uses font '{font_name}' but it could not be found")
   276→                missing_fonts.append(font_name)
   277→
   278→        return font_count, missing_fonts

   293→    def generate_sidecar_subtitle_path(
   294→        self,
   295→        subtitle: Subtitle,
   296→        base_filename: str,
   297→        output_dir: Path,
   298→        target_codec: Optional[Subtitle.Codec] = None,
   299→        source_path: Optional[Path] = None,
   300→    ) -> Path:
   301→        """Generate sidecar path: {base}.{lang}[.forced][.sdh].{ext}"""
   302→        lang_suffix = str(subtitle.language) if subtitle.language else "und"
   303→        forced_suffix = ".forced" if subtitle.forced else ""
   304→        sdh_suffix = ".sdh" if (subtitle.sdh or subtitle.cc) else ""
   305→
   306→        extension = (target_codec or subtitle.codec or Subtitle.Codec.SubRip).extension
   307→        if not target_codec and not subtitle.codec and source_path and source_path.suffix:
   308→            extension = source_path.suffix.lstrip(".")
   309→
   310→        filename = f"{base_filename}.{lang_suffix}{forced_suffix}{sdh_suffix}.{extension}"
   311→        return output_dir / filename

   313→    def output_subtitle_sidecars(
   314→        self,
   315→        subtitles: list[Subtitle],
   316→        base_filename: str,
   317→        output_dir: Path,
   318→        sidecar_format: str,
   319→        original_paths: Optional[dict[str, Path]] = None,
   320→    ) -> list[Path]:
   321→        """Output subtitles as sidecar files, converting if needed."""
   322→        created_paths: list[Path] = []
   323→        config.directories.temp.mkdir(parents=True, exist_ok=True)
   324→
   325→        for subtitle in subtitles:
   326→            source_path = subtitle.path
   327→            if sidecar_format == "original" and original_paths and subtitle.id in original_paths:
   328→                source_path = original_paths[subtitle.id]
   329→
   330→            if not source_path or not source_path.exists():
   331→                continue
   332→
   333→            # Determine target codec
   334→            if sidecar_format == "original":
   335→                target_codec = None
   336→                if source_path.suffix:
   337→                    try:
   338→                        target_codec = Subtitle.Codec.from_mime(source_path.suffix.lstrip("."))
   339→                    except ValueError:
   340→                        target_codec = None
   341→            else:
   342→                target_codec = Subtitle.Codec.from_mime(sidecar_format)
   343→
   344→            sidecar_path = self.generate_sidecar_subtitle_path(
   345→                subtitle, base_filename, output_dir, target_codec, source_path=source_path
   346→            )
   347→
   348→            # Copy or convert
   349→            if not target_codec or subtitle.codec == target_codec:
   350→                shutil.copy2(source_path, sidecar_path)
   351→            else:
   352→                # Create temp copy for conversion to preserve original
   353→                temp_path = config.directories.temp / f"sidecar_{subtitle.id}{source_path.suffix}"
   354→                shutil.copy2(source_path, temp_path)
   355→
   356→                temp_sub = Subtitle(
   357→                    subtitle.url,
   358→                    subtitle.language,
   359→                    is_original_lang=subtitle.is_original_lang,
   360→                    descriptor=subtitle.descriptor,
   361→                    codec=subtitle.codec,
   362→                    forced=subtitle.forced,
   363→                    sdh=subtitle.sdh,
   364→                    cc=subtitle.cc,
   365→                    id_=f"{subtitle.id}_sc",
   366→                )
   367→                temp_sub.path = temp_path
   368→                try:
   369→                    temp_sub.convert(target_codec, forced=True)
   370→                    if temp_sub.path and temp_sub.path.exists():
   371→                        shutil.copy2(temp_sub.path, sidecar_path)
   372→                finally:
   373→                    if temp_sub.path and temp_sub.path.exists():
   374→                        temp_sub.path.unlink(missing_ok=True)
   375→                    temp_path.unlink(missing_ok=True)
   376→
   377→            created_paths.append(sidecar_path)
   378→
   379→        return created_paths

   712→    DRM_TABLE_LOCK = Lock()
   713→    EXPORT_LOCK = Lock()

   715→
   716→    def __init__(
   717→        self,
   718→        ctx: click.Context,
   719→        no_proxy: bool,
   720→        profile: Optional[str] = None,
   721→        proxy: Optional[str] = None,
   722→        repack: bool = False,
   723→        tag: Optional[str] = None,
   724→        tmdb_id: Optional[int] = None,
   725→        imdb_id: Optional[str] = None,
   726→        animeapi_id: Optional[str] = None,
   727→        enrich: bool = False,
   728→        output_dir: Optional[Path] = None,
   729→        *_: Any,
   730→        **__: Any,
   731→    ):
   732→        if not ctx.invoked_subcommand:
   733→            raise ValueError("A subcommand to invoke was not specified, the main code cannot continue.")
   734→
   735→        self.log = logging.getLogger("download")
   736→        self.completed_files: list[Path] = []
   737→        # Subtitles skipped under --skip-subtitle-errors, recorded so an embedding caller can
   738→        # report which weren't available without parsing the console output. See SkippedSubtitle.
   739→        self.skipped_subtitles: list[SkippedSubtitle] = []
   740→        # result() catches a download-worker failure, reports it, and returns rather than
   741→        # re-raising (so the CLI still exits cleanly). Flag it so an embedding caller (the API
   742→        # worker) can tell the title did not complete instead of reading it as a success.
   743→        self.download_failed: bool = False
   744→
   745→        if not config.output_template:
   746→            raise click.ClickException(
   747→                "No 'output_template' configured in your unshackle.yaml.\n"
   748→                "Please add an 'output_template' section with movies, series, and songs/music templates.\n"
   749→                "See unshackle-example.yaml for examples."
   750→            )
   751→
   752→        self.service = Services.get_tag(ctx.invoked_subcommand)
   753→        self.vault_service = Services.get_vault_tag(self.service)
   754→        apply_service_dl_overrides(ctx, config.services.get(self.service, {}).get("dl", {}), self.log)
   755→
   756→        # Refresh locals Click bound before the overrides ran.
   757→        no_proxy = ctx.params.get("no_proxy", no_proxy)
   758→        profile = ctx.params.get("profile", profile)
   759→        proxy = ctx.params.get("proxy", proxy)
   760→        repack = ctx.params.get("repack", repack)
   761→        tag = ctx.params.get("tag", tag)
   762→        tmdb_id = ctx.params.get("tmdb_id", tmdb_id)
   763→        imdb_id = ctx.params.get("imdb_id", imdb_id)
   764→        animeapi_id = ctx.params.get("animeapi_id", animeapi_id)
   765→        enrich = ctx.params.get("enrich", enrich)
   766→        output_dir = ctx.params.get("output_dir", output_dir)
   767→
   768→        self.profile = profile
   769→        self.proxy_requested = bool(proxy)
   770→        self.tmdb_id = tmdb_id
   771→        self.imdb_id = imdb_id
   772→        self.enrich = enrich
   773→        self.animeapi_title: Optional[str] = None
   774→        self.output_dir = output_dir
   775→
   776→        if animeapi_id:
   777→            from unshackle.core.utils.animeapi import resolve_animeapi
   778→
   779→            anime_title, anime_ids = resolve_animeapi(animeapi_id)
   780→            self.animeapi_title = anime_title
   781→            if not self.tmdb_id and anime_ids.tmdb_id:
   782→                self.tmdb_id = anime_ids.tmdb_id
   783→            if not self.imdb_id and anime_ids.imdb_id:
   784→                self.imdb_id = anime_ids.imdb_id
   785→
   786→        if self.enrich and not (self.tmdb_id or self.imdb_id or self.animeapi_title):
   787→            raise click.UsageError("--enrich requires --tmdb, --imdb, or --animeapi to provide a metadata source.")
   788→
   789→        # Initialize debug logger with service name if debug logging is enabled
   790→        if config.debug or logging.root.level == logging.DEBUG:
   791→            from collections import defaultdict
   792→            from datetime import datetime
   793→
   794→            debug_log_path = config.directories.logs / config.filenames.debug_log.format_map(
   795→                defaultdict(str, service=self.service, time=datetime.now().strftime("%Y%m%d-%H%M%S"))
   796→            )
   797→            init_debug_logger(log_path=debug_log_path, enabled=True, log_keys=config.debug_keys)
   798→            self.debug_logger = get_debug_logger()
   799→
   800→            if self.debug_logger:
   801→                self.debug_logger.log(
   802→                    level="INFO",
   803→                    operation="download_init",
   804→                    message=f"Download command initialized for service {self.service}",
   805→                    service=self.service,
   806→                    context={
   807→                        "profile": profile,
   808→                        "proxy": proxy,
   809→                        "tag": tag,
   810→                        "tmdb_id": self.tmdb_id,
   811→                        "imdb_id": self.imdb_id,
   812→                        "animeapi_id": animeapi_id,
   813→                        "enrich": enrich,
   814→                        "cli_params": {
   815→                            k: v
   816→                            for k, v in ctx.params.items()
   817→                            if k
   818→                            not in [
   819→                                "profile",
   820→                                "proxy",
   821→                                "tag",
   822→                                "tmdb_id",
   823→                                "imdb_id",
   824→                                "animeapi_id",
   825→                                "enrich",
   826→                            ]
   827→                        },
   828→                    },
   829→                )
   830→
   831→                # Log binary versions for diagnostics
   832→                binary_versions = {}
   833→                for name, binary in [
   834→                    ("shaka_packager", binaries.ShakaPackager),
   835→                    ("mp4decrypt", binaries.Mp4decrypt),
   836→                    ("mkvmerge", binaries.MKVToolNix),
   837→                    ("ffmpeg", binaries.FFMPEG),
   838→                    ("ffprobe", binaries.FFProbe),
   839→                ]:
   840→                    if binary:
   841→                        version = None
   842→                        try:
   843→                            if name == "shaka_packager":
   844→                                r = subprocess.run(
   845→                                    [str(binary), "--version"], capture_output=True, text=True, timeout=5
   846→                                )
   847→                                version = (r.stdout or r.stderr or "").strip()
   848→                            elif name in ("ffmpeg", "ffprobe"):
   849→                                r = subprocess.run([str(binary), "-version"], capture_output=True, text=True, timeout=5)
   850→                                version = (r.stdout or "").split("\n")[0].strip()
   851→                            elif name == "mkvmerge":
   852→                                r = subprocess.run(
   853→                                    [str(binary), "--version"], capture_output=True, text=True, timeout=5
   854→                                )
   855→                                version = (r.stdout or "").strip()
   856→                            elif name == "mp4decrypt":
   857→                                r = subprocess.run([str(binary)], capture_output=True, text=True, timeout=5)
   858→                                output = (r.stdout or "") + (r.stderr or "")
   859→                                lines = [line.strip() for line in output.split("\n") if line.strip()]
   860→                                version = " | ".join(lines[:2]) if lines else None
   861→                        except Exception:
   862→                            version = "<error getting version>"
   863→                        binary_versions[name] = {"path": str(binary), "version": version}
   864→                    else:
   865→                        binary_versions[name] = None
   866→
   867→                self.debug_logger.log(
   868→                    level="DEBUG",
   869→                    operation="binary_versions",
   870→                    message="Binary tool versions",
   871→                    context=binary_versions,
   872→                )
   873→        else:
   874→            self.debug_logger = None
   875→
   876→        if self.profile:
   877→            self.log.info(f"Using profile: '{self.profile}'")
   878→
   879→        self.is_remote = bool(ctx.params.get("remote"))
   880→
   881→        with console.status("Loading Service Config...", spinner="dots"):
   882→            self.service_config = {}
   883→            if not self.is_remote:
   884→                try:
   885→                    service_config_path = Services.get_path(self.service) / config.filenames.config
   886→                    if service_config_path.exists():
   887→                        self.service_config = yaml.safe_load(service_config_path.read_text(encoding="utf8"))
   888→                        self.log.info("Service Config loaded")
   889→                        # log key names only -- the full config carries service certificates,
   890→                        # device fingerprints and endpoints that bloat the log and may be sensitive
   891→                        log_event(
   892→                            "load_service_config",
   893→                            level="DEBUG",
   894→                            service=self.service,
   895→                            context={
   896→                                "config_path": str(service_config_path),
   897→                                "config_keys": sorted(self.service_config or []),
   898→                            },
   899→                        )
   900→                except KeyError:
   901→                    pass
   902→            merge_dict(config.services.get(self.service), self.service_config)
   903→
   904→        service_config = config.services.get(self.service, {})
   905→        if service_config:
   906→            reserved_keys = {
   907→                "profiles",
   908→                "api_key",
   909→                "certificate",
   910→                "api_endpoint",
   911→                "region",
   912→                "device",
   913→                "endpoints",
   914→                "client",
   915→                "dl",
   916→            }
   917→
   918→            for config_key, override_value in service_config.items():
   919→                if config_key in reserved_keys or not isinstance(override_value, dict):
   920→                    continue
   921→
   922→                if hasattr(config, config_key):
   923→                    current_config = getattr(config, config_key, {})
   924→
   925→                    if isinstance(current_config, dict):
   926→                        merged_config = deepcopy(current_config)
   927→                        merge_dict(override_value, merged_config)
   928→                        setattr(config, config_key, merged_config)
   929→
   930→                        self.log.debug(
   931→                            f"Applied service-specific '{config_key}' overrides for {self.service}: {override_value}"
   932→                        )
   933→
   934→        cdm_only = ctx.params.get("cdm_only")
   935→
   936→        if cdm_only:
   937→            self.vaults = Vaults(self.vault_service)
   938→            self.log.info("CDM-only mode: Skipping vault loading")
   939→            log_event(
   940→                "vault_loading_skipped",
   941→                level="INFO",
   942→                service=self.service,
   943→                context={"reason": "cdm_only flag set"},
   944→            )
   945→        else:
   946→            with console.status("Loading Key Vaults...", spinner="dots"):
   947→                self.vaults = Vaults(self.vault_service)
   948→                total_vaults = len(config.key_vaults)
   949→                failed_vaults = []
   950→
   951→                for vault in config.key_vaults:
   952→                    vault_type = vault["type"]
   953→                    vault_name = vault.get("name", vault_type)
   954→                    vault_copy = vault.copy()
   955→                    del vault_copy["type"]
   956→
   957→                    if vault_type.lower() == "api" and "decrypt_labs" in vault_name.lower():
   958→                        if "token" not in vault_copy or not vault_copy["token"]:
   959→                            if config.decrypt_labs_api_key:
   960→                                vault_copy["token"] = config.decrypt_labs_api_key
   961→                            else:
   962→                                self.log.warning(
   963→                                    f"No token provided for DecryptLabs vault '{vault_name}' and no global "
   964→                                    "decrypt_labs_api_key configured"
   965→                                )
   966→
   967→                    if vault_type.lower() == "sqlite":
   968→                        try:
   969→                            self.vaults.load_critical(vault_type, **vault_copy)
   970→                            self.log.debug(f"Successfully loaded vault: {vault_name} ({vault_type})")
   971→                        except Exception as e:
   972→                            self.log.error(f"vault failure: {vault_name} ({vault_type}) - {e}")
   973→                            raise
   974→                    else:
   975→                        # Other vaults (MySQL, HTTP, API) - soft fail
   976→                        if not self.vaults.load(vault_type, **vault_copy):
   977→                            failed_vaults.append(vault_name)
   978→                            self.log.debug(f"Failed to load vault: {vault_name} ({vault_type})")
   979→                        else:
   980→                            self.log.debug(f"Successfully loaded vault: {vault_name} ({vault_type})")
   981→
   982→                loaded_count = len(self.vaults)
   983→                if failed_vaults:
   984→                    self.log.warning(f"Failed to load {len(failed_vaults)} vault(s): {', '.join(failed_vaults)}")
   985→                self.log.info(f"Loaded {loaded_count}/{total_vaults} Vaults")
   986→
   987→                # Debug: Show detailed vault status
   988→                if loaded_count > 0:
   989→                    vault_names = [vault.name for vault in self.vaults]
   990→                    self.log.debug(f"Active vaults: {', '.join(vault_names)}")
   991→                else:
   992→                    self.log.debug("No vaults are currently active")
   993→
   994→        with console.status("Loading DRM CDM...", spinner="dots"):
   995→            try:
   996→                self.cdm = self.get_cdm(self.service, self.profile)
   997→            except ValueError as e:
   998→                self.log.error(f"Failed to load CDM, {e}")
   999→                if self.debug_logger:
  1000→                    self.debug_logger.log_error("load_cdm", e, service=self.service)
  1001→                sys.exit(1)
  1002→
  1003→            if self.cdm:
  1004→                cdm_info = {}
  1005→                if isinstance(self.cdm, DecryptLabsRemoteCDM):
  1006→                    drm_type = "PlayReady" if self.cdm.is_playready else "Widevine"
  1007→                    self.log.info(f"Loaded {drm_type} Remote CDM: DecryptLabs (L{self.cdm.security_level})")
  1008→                    cdm_info = {"type": "DecryptLabs", "drm_type": drm_type, "security_level": self.cdm.security_level}
  1009→                elif hasattr(self.cdm, "device_type") and self.cdm.device_type.name in ["ANDROID", "CHROME"]:
  1010→                    self.log.info(f"Loaded Widevine CDM: {self.cdm.system_id} (L{self.cdm.security_level})")
  1011→                    cdm_info = {
  1012→                        "type": "Widevine",
  1013→                        "system_id": self.cdm.system_id,
  1014→                        "security_level": self.cdm.security_level,
  1015→                        "device_type": self.cdm.device_type.name,
  1016→                    }
  1017→                else:
  1018→                    # Handle both local PlayReady CDM and RemoteCdm (which has certificate_chain=None)
  1019→                    is_remote = self.cdm.certificate_chain is None and hasattr(self.cdm, "device_name")
  1020→                    if is_remote:
  1021→                        cdm_name = self.cdm.device_name
  1022→                        self.log.info(f"Loaded PlayReady Remote CDM: {cdm_name} (L{self.cdm.security_level})")
  1023→                    else:
  1024→                        cdm_name = self.cdm.certificate_chain.get_name() if self.cdm.certificate_chain else "Unknown"
  1025→                        self.log.info(f"Loaded PlayReady CDM: {cdm_name} (L{self.cdm.security_level})")
  1026→                    cdm_info = {
  1027→                        "type": "PlayReady",
  1028→                        "certificate": cdm_name,
  1029→                        "security_level": self.cdm.security_level,
  1030→                    }
  1031→
  1032→                if cdm_info:
  1033→                    log_event("load_cdm", level="INFO", service=self.service, context={"cdm": cdm_info})
  1034→
  1035→        self.proxy_providers = []
  1036→        if no_proxy:
  1037→            ctx.params["proxy"] = None
  1038→        else:
  1039→            with console.status("Loading Proxy Providers...", spinner="dots"):
